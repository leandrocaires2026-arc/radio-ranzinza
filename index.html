function ligarSomOuvinte() { somLiberadoOuvinte = true; const triggerBox = document.getElementById('playTriggerBox'); if (triggerBox) triggerBox.style.display = 'none'; // Força o início imediato do áudio ativo sem travar nas promessas tocarItemAtual(); } function tocarItemAtual() { const item = playlistGlobal[indiceAtualPlaylist]; const audio1 = document.getElementById('audioPlayer1'); const audio2 = document.getElementById('audioPlayer2'); const nowPlayingText = document.getElementById('now-playing'); if (!audio1 || !audio2 || !nowPlayingText) return; const activeAudio = playerAtivo === 1 ? audio1 : audio2; const nextAudio = playerAtivo === 1 ? audio2 : audio1; if (item.type === 'jingle') { nowPlayingText.innerHTML = `No ar agora: ${item.title}`; audio1.pause(); audio2.pause(); falarJingleNarrador(); } else { nowPlayingText.innerHTML = `No ar agora: ${item.title}`; // Configura o próximo player nextAudio.src = item.url; nextAudio.load(); if (offsetSegundosInicial > 0) { nextAudio.onloadedmetadata = () => { if (offsetSegundosInicial < nextAudio.duration) { nextAudio.currentTime = offsetSegundosInicial; } }; } // Se o ouvinte já liberou o som, disparamos o play de forma limpa e imediata if (somLiberadoOuvinte) { nextAudio.play().then(() => { activeAudio.pause(); activeAudio.currentTime = 0; playerAtivo = playerAtivo === 1 ? 2 : 1; }).catch(err => { console.log("Autoplay retido pelo navegador, tentando novamente...", err); // Fallback caso o navegador exija mais um gesto setTimeout(() => nextAudio.play().catch(e => console.log("Erro final de play:", e)), 500); }); } else { activeAudio.pause(); playerAtivo = playerAtivo === 1 ? 2 : 1; } offsetSegundosInicial = 0; } // Garante a transição automática para a próxima faixa ao terminar nextAudio.onended = () => { tocarProximaFaixa(); }; }
